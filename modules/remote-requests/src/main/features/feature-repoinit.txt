# Everything a remote peer has ever caused to be created, kept apart from the content the platform
# itself owns. Nothing is granted on it to anyone: the endpoint is the only way in, and it writes as
# the service user below.
create path (sling:Folder) /remote

# The write that follows a verified token. Its rights stop at /remote, so a flaw in the endpoint's own
# name checking still cannot reach content that matters.
create service user iap-remote-requests

set ACL for iap-remote-requests
    allow   jcr:read,jcr:write   on /remote
end
