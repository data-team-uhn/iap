create path (iap:LinkTypesHomepage) /LinkTypes

# Every user needs to know what the links placed on the resources they can see mean
set ACL for everyone
    allow   jcr:read    on /LinkTypes
end

# The service user completing backlinks and creating link containers; needs to
# write anywhere links can be placed, and to check out versionable resources
# when adding their link container
create service user iap-links

set ACL for iap-links
    allow   jcr:read,rep:write,jcr:versionManagement    on /
end
