create path (app:LoginPage) /login
create path (app:Configuration) /libs/iap/conf/LoginPage

# The login page is readable without a session. Sling also has to let the request through
# without one, which the `-/login` entry of `sling.auth.requirements` does, in
# packaging/slingfeature/src/main/features/core/sling-configuration.json.
set ACL for everyone
    allow   jcr:read    on /login
end

# The login page's texts, which reach the page as <meta> tags. `default` leaves a value a
# deployment has already set as it is. The text introducing the platform is initial content,
# in SLING-INF/content/libs/iap/conf/LoginPage.json.
set properties on /libs/iap/conf/LoginPage
  default tagline to "Where research moves forward"
  default signInLabel to "Sign in"
  default signInHeading to "Continue with institutional credentials"
end
