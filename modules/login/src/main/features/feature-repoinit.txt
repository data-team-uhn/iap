create path (app:LoginPage) /login
create path (app:Configuration) /libs/iap/conf/LoginPage

# The login page must be readable without a session. This is the JCR half of the access
# setup; the HTTP half is the -/login exemption in sling.auth.requirements
# (packaging/slingfeature/src/main/features/core/sling-configuration.json). Both are needed.
set ACL for everyone
    allow   jcr:read    on /login
end

# The login page texts, delivered to the page as meta tags like the rest of the branding;
# deployments override them without code changes. The intro text (what the platform does) is
# not seeded here; the pilot wording ships as initial content instead
# (SLING-INF/content/libs/iap/conf/LoginPage.json). See the module README for the conf
# reference and for the wording to switch to once several institutions participate.
set properties on /libs/iap/conf/LoginPage
  default tagline to "Where research moves forward"
  default signInLabel to "Sign in"
  default signInHeading to "Continue with institutional credentials"
end
