create path (mail:CaughtMailHomepage) /CaughtMail

# The node that answers, to anyone, whether mail is being caught. It is created here, not shipped as initial
# content, because repoinit runs before initial content loads and cannot set an ACL on a path that does not exist.
create path (sling:Folder) /libs/iap
create path (mail:CatcherStatus) /libs/iap/mail-catcher

# The user the catcher files messages as. It may write under /CaughtMail, and nowhere else.
create service user iap-email-catcher

# Also created by other modules. Creating it here keeps the grant below from depending on their order.
create group iap-administrators

set ACL for iap-email-catcher
    allow   jcr:read,jcr:addChildNodes,jcr:modifyProperties,jcr:nodeTypeManagement    on /CaughtMail
end

# Administrators only, since a caught message may hold private information.
set ACL for iap-administrators
    allow   jcr:read    on /CaughtMail
end

# Whether mail is being caught, for everyone. /libs is exempt from sling.auth.requirements, so this answers
# before sign-in too. frontend-commons grants everyone read on all of /libs/iap, and this grant does not rely
# on it.
set ACL for everyone
    allow   jcr:read    on /libs/iap/mail-catcher
end
