# Where deleted resources are kept until restored or purged. Granted to
# nobody, so ordinary users cannot read it and their queries never return
# archived content. Administrators bypass access control.
create path (del:Archive) /Archive

# The deletion service's user. It finds the references to a resource
# whoever can see them, and moves content in and out of the archive anywhere
# in the tree. The service checks the requesting user's own permissions first.
create service user iap-deletion

set ACL for iap-deletion
    allow   jcr:all     on /
end
