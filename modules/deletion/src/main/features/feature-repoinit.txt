# Where deleted resources are kept until restored or purged. Deliberately not
# granted to anyone: regular users must not see deleted content, and queries
# must not return it; administrators bypass access control.
create path (del:Archive) /Archive

# The service user performing the impact analysis and all the changes: it
# scans references regardless of who can see them, writes the archive, and
# moves content in and out of it, anywhere in the tree. Security is enforced
# beforehand against the requesting user's own session.
create service user iap-deletion

set ACL for iap-deletion
    allow   jcr:all     on /
end
