create path (hist:Log) /History

# Nobody is granted read on this store, neither `everyone` nor the people whose work it describes. This service user
# reads it, and serves the person asking what they may see.
create service user iap-history

# No jcr:removeNode or jcr:removeChildNodes anywhere below: the record is append-only, and nobody holding these grants
# can delete from it.
#
# This user only maintains the prefix-tree buckets. An action is written in the session of whoever makes the change it
# describes, so that the two commit together.
set ACL for iap-history
    allow   jcr:read,jcr:addChildNodes,jcr:modifyProperties,jcr:nodeTypeManagement    on /History
end

# Whoever records an action writes it in their own session, so the privilege has to be theirs. A module that records
# joins this group in its own repoinit.
create group iap-history-writers

set ACL for iap-history-writers
    allow   jcr:read,jcr:addChildNodes,jcr:modifyProperties,jcr:nodeTypeManagement    on /History
end
