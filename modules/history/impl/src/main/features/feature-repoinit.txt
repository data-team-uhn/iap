create path (hist:Log) /History

# No person is granted read on this store, neither `everyone` nor the people whose work it describes.
create service user iap-history

# The store's own user, and the one user meant to read it, for the service that serves the history.
set ACL for iap-history
    allow   jcr:read    on /History
end

# An action is written by the commit it records, as the user who commits, so the privilege has to be theirs. A module
# that records, or attaches snapshots, joins this group in its own repoinit.
#
# No jcr:removeNode or jcr:removeChildNodes: nobody holding these grants can delete an action or an entry.
# jcr:modifyProperties lets them change one, which attaching a snapshot afterwards needs.
create group iap-history-writers

set ACL for iap-history-writers
    allow   jcr:read,jcr:addChildNodes,jcr:modifyProperties,jcr:nodeTypeManagement    on /History
end
