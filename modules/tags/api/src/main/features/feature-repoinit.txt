create path (tag:Homepage) /Tags

# Every user may read what the tags they see mean
set ACL for everyone
    allow   jcr:read    on /Tags
end

# The tag manager reads the definitions with its own service user, so a caller needs no session of its own. It may
# only read them. Every operation on tagged content goes through the caller's own session.
create service user iap-tags

set ACL for iap-tags
    allow   jcr:read    on /Tags
end

# The tag repair marks stale content anywhere in the repository for recomputation, on a schedule or on request. It
# may read anything and set properties, but never create or remove a node. Oak checks the propagation editor's writes
# against this same user, and they too are only properties.
create service user iap-tag-repair

set ACL for iap-tag-repair
    allow   jcr:read,jcr:modifyProperties    on /
end
