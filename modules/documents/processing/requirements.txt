# Runtime dependencies of the Docling image, pinned.
#
# The Dockerfile used to install "docling>=2.115" with no upper bound, so the same commit
# built on two days could produce two different images. That is worse here than in most
# projects: the next layer runs docling's download_models(), so the *model weights* baked
# into the image change with the library version, and CI never builds this image
# (pull-request-build.yml has no -Pdocker), so nothing catches the drift before a deploy.
#
# Bump these deliberately, in a commit of their own, and rebuild the image to confirm:
#
#   docker build -t iap/docling:latest modules/documents/processing
#   docker run --rm --entrypoint python iap/docling:latest \
#     -c "import importlib.metadata as m; print(m.version('docling'))"
#
# These pin the direct dependencies only. Their transitive graph (torch, transformers and
# the rest of the Docling stack) still floats, which is the remaining reproducibility gap;
# closing it means a fully resolved, hashed lock file generated with pip-compile against the
# image's own platform, since several of these ship platform-specific wheels.
docling==2.120.1
pypdf==6.16.1
# pypdf's AES crypt provider. Signed / permission-restricted PDFs are often AES-encrypted
# with an empty user password; without this, reading the page tree raises
# DependencyError: cryptography>=3.1 is required for AES algorithm.
cryptography==46.0.7
psutil==7.2.2
chunkweaver==0.4.0
