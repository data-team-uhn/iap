create path (notif:Homepage) /Notifications

# The channel's own writer: it files notifications, and it grants each recipient read on their own, which is why
# it may manage access control. No jcr:removeNode, deliberately: nothing stored here is ever deleted by the
# platform, and a service user that cannot delete makes that a property of the deployment rather than a promise
# made in code.
create service user iap-stored-notifications
set ACL for iap-stored-notifications
    allow   jcr:read,jcr:addChildNodes,jcr:modifyProperties,jcr:nodeTypeManagement,jcr:readAccessControl,jcr:modifyAccessControl    on /Notifications
end

# Enough for a signed-in reader to traverse down to their own notifications, and not one node more: the homepage
# and the prefix-tree buckets (plain folders) are readable, while each notif:Notification below them is invisible
# until the delivery grants its one recipient read on it. A listing therefore never needs filtering code - the
# repository already answered who may see what.
set ACL on /Notifications
    allow jcr:read for everyone restriction(rep:ntNames,notif:Homepage,sling:Folder)
end
