create path (link:TypesHomepage) /LinkTypes

# Everybody may read the link definitions, which say what each link means
set ACL for everyone
    allow   jcr:read    on /LinkTypes
end

# Adds the missing link containers, and the reverse links a user may not add.
# It writes wherever links go, and checks out a checked-in resource to add its
# container.
create service user iap-links

set ACL for iap-links
    allow   jcr:read,rep:write,jcr:versionManagement    on /
end

# Reads the link definitions for the link manager, whose callers may have no
# session of their own. It can read nothing else.
create service user iap-link-types

set ACL for iap-link-types
    allow   jcr:read    on /LinkTypes
end
