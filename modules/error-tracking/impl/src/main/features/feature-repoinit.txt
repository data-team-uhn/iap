# A recorded error quotes whatever the failing code was working on, so nobody is granted read access here, not even
# `everyone` on the homepage. Only this service user, service users that may read the whole repository, and the
# repository superuser can read it.
create path (err:LoggedErrorsHomepage) /LoggedErrors

# The user that records the errors and reads them for the status report
create service user iap-error-tracking

# No jcr:removeNode and no jcr:removeChildNodes: nothing recorded here is ever deleted
set ACL for iap-error-tracking
    allow   jcr:read,jcr:addChildNodes,jcr:modifyProperties,jcr:nodeTypeManagement    on /LoggedErrors
end
