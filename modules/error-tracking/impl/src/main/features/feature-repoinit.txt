create path (err:LoggedErrorsHomepage) /LoggedErrors

# Recorded errors quote whatever the failing code was working on, so no read access is granted here to anybody: what
# can reach them is this service user, the platform-wide readers Sling installs, and administrators. In particular
# `everyone` is deliberately not granted anything, not even on the homepage node.
create service user iap-error-tracking

# No jcr:removeNode and no jcr:removeChildNodes, deliberately: nothing recorded here is ever deleted, and a service
# user that cannot delete makes that a property of the deployment rather than a promise made in code
set ACL for iap-error-tracking
    allow   jcr:read,jcr:addChildNodes,jcr:modifyProperties,jcr:nodeTypeManagement    on /LoggedErrors
end
