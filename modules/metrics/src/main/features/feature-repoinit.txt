create path (metric:Homepage) /Metrics

# The user that reads and writes the metrics. Nobody else is granted access to the metrics' nodes. People read the
# metrics through /Metrics.json and the status report, which apply each metric's access level.
create service user iap-metrics with path system/iap

set ACL for iap-metrics
    allow jcr:read,jcr:write,jcr:nodeTypeManagement on /Metrics
end

# Everyone may read /Metrics itself, so that /Metrics.json resolves, but not the metrics under it
set ACL for everyone
    allow jcr:read on /Metrics restriction(rep:ntNames,metric:Homepage)
end
